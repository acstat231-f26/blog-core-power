# -----------------------------------------------------------------------------
# This is the code for scraping 10-Q R&D expenses from the SEC website's HTML 
# files for Microsoft, Google, Meta, Apple, Nvidia, and Tesla
# -----------------------------------------------------------------------------

# packages
library(tidyverse)
library(rvest)
library(robotstxt)
library(httr)

# Import dataset 
# Variable Explanations:
# COMP: Company Name
# FIL-D: Filing Date
# REP-D: Reporting For Date
# HTML: HTML link

# find data in sibling folder
ten_Qs <- readxl::read_xlsx("../data/10q-links.xlsx")

# Compartmentalize each variable
# Microsoft
microsoft <- ten_Qs |>
  filter(COMP == "Microsoft")

# Google
google <- ten_Qs |>
  filter(COMP == "Google")

# Meta
meta <- ten_Qs |>
  filter(COMP == "Meta")

# Apple
apple <- ten_Qs |>
  filter(COMP == "Apple")

# Nvidia
nvidia <- ten_Qs |>
  filter(COMP == "Nvidia")

# Tesla
tesla <- ten_Qs |>
  filter(COMP == "Tesla")

# Create SEC session
my_ua <- "AmherstCollege alxu27@amherst.edu"
sec_session <- session("https://www.sec.gov", user_agent(my_ua))

# Add empty column to store values
ten_Qs <- ten_Qs |>
  mutate(RD = NA)

# -------------------------------Scraping---------------------------------------
# Microsoft
for (i in 1:4) {
  ten_Qs$RD[i] <- session_jump_to(sec_session, microsoft$HTML[i]) |>
    read_html() |>
    html_elements("table") |>
    purrr::pluck(7) |>
    html_table() |>
    filter(X1 == "Research and development") |>
    select(X4)
}

# Google
for (i in 1:2) {

    html_elements("table") |>

    select(X10)
}

    html_elements("table") |>

# -----------------------------------------------------------------------------
# This is the code for importing and wrangling stock prices for all 6 companies 
# by using the quantmod package
# Because the 10-Q reporting time frames are different for each company, we
# used a standardized, informative time frame (2025-03-01 to 2026-11-01) that 
# captured all 4 10-Qs for each company 
# -----------------------------------------------------------------------------

# packages
library(tidyverse)
library(quantmod)

# Microsoft
microsoft_stock <- getSymbols("MSFT", src = "yahoo", 
                              from = "2025-03-01", to = "2026-11-01") |>
  data.frame(Date = index(MSFT), coredata(MSFT)) |>
  select(Date, MSFT.Close) |>
  rename(Close = MSFT.Close) |>
  mutate(COMP = "MSFT")

# Alphabet
alphabet_stock <- getSymbols("GOOG", src = "yahoo", 
                             from = "2025-03-01", to = "2026-11-01") |>
  data.frame(Date = index(GOOG), coredata(GOOG)) |>
  select(Date, GOOG.Close) |>
  rename(Close = GOOG.Close) |>
  mutate(COMP = "GOOG")

# Meta
meta_stock <- getSymbols("META", src = "yahoo", 
                         from = "2025-03-01", to = "2026-11-01") |>
  data.frame(Date = index(META), coredata(META)) |>
  select(Date, META.Close) |>
  rename(Close = META.Close) |>
  mutate(COMP = "META")

# Apple
apple_stock <- getSymbols("AAPL", src = "yahoo", 
                          from = "2025-03-01", to = "2026-11-01") |>
  data.frame(Date = index(AAPL), coredata(AAPL)) |>
  select(Date, AAPL.Close) |>
  rename(Close = AAPL.Close) |>
  mutate(COMP = "AAPL")

# Nvidia
nvidia_stock <- getSymbols("NVDA", src = "yahoo", 
                           from = "2025-03-01", to = "2026-11-01") |>
  data.frame(Date = index(NVDA), coredata(NVDA)) |>
  select(Date, NVDA.Close) |>
  rename(Close = NVDA.Close) |>
  mutate(COMP = "NVDA")

# Tesla
tesla_stock <- getSymbols("TSLA", src = "yahoo", 
                          from = "2025-03-01", to = "2026-11-01") |>
  data.frame(Date = index(TSLA), coredata(TSLA)) |>
  select(Date, TSLA.Close) |>   
  rename(Close = TSLA.Close) |>
  mutate(COMP = "TSLA")

# Binding rows to consolidate all stock price points into one data set in 
# preparation for faceted time series visualization
stock_price_joined <- bind_rows(microsoft_stock, alphabet_stock, meta_stock, 
                                apple_stock, nvidia_stock, tesla_stock)

write_csv(stock_price_joined, "../data/stock_price_joined.csv")

write_csv(microsoft_stock, "../data/microsoft_stock.csv")
write_csv(alphabet_stock, "../data/alphabet_stock.csv")
write_csv(meta_stock, "../data/meta_stock.csv")
write_csv(apple_stock, "../data/apple_stock.csv")
write_csv(nvidia_stock, "../data/nvidia_stock.csv")
write_csv(tesla_stock, "../data/tesla_stock.csv")

# -----------------------------------------------------------------------------
# This is the code for scraping 10-Qs' MD&A text from the SEC website's HTML 
# files for Microsoft, Google, Meta, Apple, Nvidia, and Tesla
# -----------------------------------------------------------------------------

# packages
library(tidyverse)
library(rvest)
library(robotstxt)
library(httr)
library(glue)

ten_Qs_text <- readxl::read_xlsx("../data/10q-links.xlsx")

# Compartmentalize each variable
# Microsoft
microsoft <- ten_Qs_text |>
  filter(COMP == "Microsoft")

# Google
google <- ten_Qs_text |>
  filter(COMP == "Google")

# Meta
meta <- ten_Qs_text |>
  filter(COMP == "Meta")

# Apple
apple <- ten_Qs_text |>
  filter(COMP == "Apple")

# Nvidia
nvidia <- ten_Qs_text |>
  filter(COMP == "Nvidia")

# Tesla
tesla <- ten_Qs_text |>
  filter(COMP == "Tesla")

# Create sec session
my_ua <- "AmherstCollege yjang27@amherst.edu"
sec_session <- session("https://www.sec.gov", user_agent(my_ua))

# Add empty column to store values
ten_Qs_text_scraped <- ten_Qs_text |>
  mutate(text = NA)

# -------------------------------Scraping---------------------------------------
# Microsoft
for (i in 1:4) {
  all_text_microsoft <- session_jump_to(sec_session, microsoft$HTML[i]) |>
    read_html() |>
    html_elements("span") |>
    html_text2()
  
  detect_1 <- which(str_detect(all_text_microsoft, "ITEM 2. MANAGEMENT’S"))
  
  detect_2 <- which(str_detect(all_text_microsoft, "ITEM 3. QUANTITATIVE"))
  
  scraped_text_microsoft <- all_text_microsoft[detect_1:detect_2]
  
  ten_Qs_text_scraped$text[i] <- paste(scraped_text_microsoft, collapse = "/n") |>
    str_replace_all("/n", " ")
}

# Google
for (i in 1:4) {
  all_text_google <- session_jump_to(sec_session, google$HTML[i]) |>
    read_html() |>
    html_elements("span") |>
    html_text2()
  
  detect_1 <- which(str_detect(all_text_google, "MANAGEMENT’S DISCUSSION AND ANALYSIS OF FINANCIAL CONDITION AND RESULTS OF OPERATIONS"))
  
  detect_2 <- which(str_detect(all_text_google, "QUANTITATIVE AND QUALITATIVE DISCLOSURES ABOUT MARKET RISK"))
  
  scraped_text_google <- all_text_google[detect_1:detect_2]
  
  ten_Qs_text_scraped$text[i+4] <- paste(scraped_text_google, collapse = "/n") |>
    str_replace_all("/n", " ")
}

# Meta
for (i in 1:4) {

    read_html() |>
    html_elements("span") |>
    html_text2()
